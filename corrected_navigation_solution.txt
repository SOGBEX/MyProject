<script type="text/javascript">
// Функция для корректного позиционирования внизу
function positionCustomSection() {
    var container = document.querySelector('.ms-quicklaunchouter');
    var customSection = document.getElementById('customQuestionSection');
    
    if (container && customSection) {
        var containerHeight = container.offsetHeight;
        var sectionHeight = customSection.offsetHeight;
        
        // Устанавливаем позицию внизу контейнера
        customSection.style.top = (containerHeight - sectionHeight - 10) + 'px';
    }
}

// Выполняем позиционирование после загрузки страницы
if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', positionCustomSection);
} else {
    positionCustomSection();
}

// Также выполняем при изменении размера окна
window.addEventListener('resize', positionCustomSection);
</script>
