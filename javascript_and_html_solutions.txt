ВАРИАНТ 1: JavaScript + изменение HTML inline стилей

1. В вашем HTML найдите эти строки и измените их:

НАЙТИ:
style="overflow: auto;height: 400px;width: 150px; "

ЗАМЕНИТЬ НА:
style="overflow: auto;height: 400px;width: 220px; "

НАЙТИ:
style="overflow: auto;height: 400px;width: 155px; "

ЗАМЕНИТЬ НА:
style="overflow: auto;height: 400px;width: 220px; "

2. Добавьте этот JavaScript код ПЕРЕД закрывающим тегом </body> или в head:

<script type="text/javascript">
function forceExpandNavigation() {
    try {
        // Расширяем главную ячейку
        var leftCell = document.getElementById('LeftNavigationAreaCell');
        if (leftCell) {
            leftCell.style.width = 'auto';
            leftCell.style.minWidth = '230px';
        }
        
        // Расширяем все таблицы навигации
        var navTables = document.querySelectorAll('.ms-nav, .ms-navframe');
        navTables.forEach(function(table) {
            table.style.width = 'auto';
            table.style.minWidth = '230px';
        });
        
        // Расширяем контейнеры быстрого запуска
        var quickLaunch = document.querySelectorAll('.ms-quicklaunchouter, .ms-quickLaunch');
        quickLaunch.forEach(function(container) {
            container.style.width = 'auto';
            container.style.minWidth = '220px';
            container.style.overflow = 'visible';
        });
        
        // Расширяем TreeView контейнеры
        var treeViews = document.querySelectorAll('#TreeViewRememberScroll, #TreeViewRememberScrollV4, .ms-treeviewouter');
        treeViews.forEach(function(tree) {
            tree.style.width = '220px';
            tree.style.minWidth = '220px';
        });
        
        // Расширяем все TD с width="100%"
        var tdElements = document.querySelectorAll('td[width="100%"]');
        tdElements.forEach(function(td) {
            td.style.width = 'auto';
            td.style.minWidth = '230px';
        });
        
        console.log('Navigation containers expanded successfully');
    } catch (e) {
        console.error('Error expanding navigation:', e);
    }
}

// Выполняем несколько раз для надежности
document.addEventListener('DOMContentLoaded', forceExpandNavigation);
window.addEventListener('load', forceExpandNavigation);
setTimeout(forceExpandNavigation, 500);
setTimeout(forceExpandNavigation, 1000);
</script>
